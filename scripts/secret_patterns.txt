# Credential-shaped patterns, one extended regular expression per line.
# Used by scripts/check_secrets.sh (grep -E, no inline flags - keep it portable).
#
# A match is a *suspicion*, not a verdict: the script only blocks the commit
# and tells you what it saw. Placeholders like TUYA_PASS=... are deliberately
# not matched (values must look real: 8+ real characters).

# Tuya device local key (the LAN control credential)
local_key"?[[:space:]]*:[[:space:]]*"[A-Za-z0-9]{16,}
"local_key"[[:space:]]*:[[:space:]]*"[A-Za-z0-9]{16,}

# Credentials in .env-style files
TUYA_PASS[[:space:]]*=[[:space:]]*["']?[A-Za-z0-9!@#$%^&*._-]{8,}
TUYA_CLIENT_SECRET[[:space:]]*=[[:space:]]*["']?[A-Za-z0-9!@#$%^&*._-]{8,}

# Password / token fields in JSON
[Pp]assword"?[[:space:]]*[:=][[:space:]]*"[^"$]{6,}
"access_token"[[:space:]]*:[[:space:]]*"[A-Za-z0-9_.-]{20,}
"refresh_token"[[:space:]]*:[[:space:]]*"[A-Za-z0-9_.-]{20,}

# HTTP headers
[Aa]uthorization:[[:space:]]*[Bb]earer[[:space:]]+[A-Za-z0-9_.-]{20,}
